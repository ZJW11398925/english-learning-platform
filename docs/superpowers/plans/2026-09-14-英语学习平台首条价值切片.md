# 英语学习平台 · 首条价值切片 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付一个手机浏览器可用的学习闭环——拍照识物取词 → 跟读 → 自己造句 → 拿到结构化反馈 → 该词在新场景中复现——用于回答"成人愿不愿意付造句这份主动产出成本"。

**Architecture:** 一个 Node 最小代理服务（托管静态资源 + 转发视觉/文本模型调用，密钥只存在于服务端），前端为一个单页状态机（七个状态，无框架，原生 ES 模块）。纯逻辑单元（`pickWord` / `frameQC` / `validateFeedback` / `scheduler` / `eventSchema`）与外部依赖（相机、转写、模型）严格分离，前者全部可单测，后者不做 mock 测试。

**Tech Stack:** Node 22+（内置 `http`/`fetch`）、原生 ES 模块前端、IndexedDB + localStorage、`node:test` 加 `node:assert/strict` 做纯逻辑单测（零测试框架依赖）、cloudflared 内网穿透提供 HTTPS。

## Global Constraints

- **密钥绝不进客户端**：所有模型调用经代理服务转发；`.env` 已被 `.gitignore` 排除，只提交 `.env.example`。
- **HTTPS 是硬要求**：`getUserMedia` 只在安全上下文可用（`localhost` 或 HTTPS 域名；`http://` + 局域网 IP 不可用）。
- **失败不得静默降级为成功**：每个故障只退到设计文档 §5.1 枚举的对应档位，并落对应标签。
- **`uncertain` 必须如实呈现**：不得把"拿不准"伪装成判定；`uncertain` 单独统计，不计入通过率。
- **不做音素级音准评分**（首版显式妥协）：只判"有没有说出目标词"。
- **识物不做自动连拍兜底**：以保住 `retry_rate` 的可测性。
- **图片存储上限**：IndexedDB，512px 长边，仅保留最近 20 个词；词与事件记录永久保留。
- **固定间隔**：1 天 / 3 天 / 7 天，完成 7 天档后转 `maintained`，不再主动推送。
- **自评测难只记录，不影响间隔**（首版不做自适应算法）。
- 提交信息用中文，一次任务一次提交，遵循 `feat:` / `test:` / `chore:` 前缀。
- 测试命令：单文件用 `node --test tests/<file>.test.mjs`；**全量用裸 `node --test`**（Windows 上 `node --test tests/` 与 `node --test tests` 都会把目录当模块路径解析而失败，已实测）。

// web/units/event-log.mjs
export const EVENT_TYPES = [
  'session_start',
  'frame_rejected',
  'blocked_permission',
  'recognize_ok',
  'recognize_failed',
  'word_shown',
  'reading_done',
  'skipped_reading',
  'speech_unsupported',
  'compose_submitted',
  'compose_rewrite',
  'feedback_ok',
  'feedback_pending',
  'uncertain',
  'recurrence_scene',
  'recurrence_manual',
  'storage_full',
];

const REQUIRED = ['ts', 'type', 'wordId', 'sessionId', 'payload'];

export function validateEvent(e) {
  const errors = [];
  if (e === null || typeof e !== 'object') return { ok: false, errors: ['event 必须是对象'] };
  for (const k of REQUIRED) {
    if (!(k in e)) errors.push(`缺少字段 ${k}`);
  }
  if ('type' in e && !EVENT_TYPES.includes(e.type)) {
    errors.push(`未登记的 type: ${String(e.type)}`);
  }
  if ('ts' in e && !Number.isFinite(e.ts)) errors.push('ts 必须是有限数字');
  if ('payload' in e && (e.payload === null || typeof e.payload !== 'object')) {
    errors.push('payload 必须是对象');
  }
  return { ok: errors.length === 0, errors };
}

export function recordEvent(store, type, { sessionId, wordId = null, ...payload }, now = Date.now) {
  const e = { ts: now(), type, wordId, sessionId, payload };
  const v = validateEvent(e);
  if (!v.ok) throw new Error(`非法事件: ${v.errors.join('; ')}`);
  store.appendEvent(e);
  return e;
}
```

// web/units/store.mjs
const EVENTS_KEY = 'elp.events';
const WORDS_KEY = 'elp.words';
const META_KEY = 'elp.meta';
const DB_NAME = 'elp-images';
const STORE_NAME = 'images';

export function createStore({ localStorage, indexedDB }) {
  const readJson = (k, fallback) => {
    const raw = localStorage.getItem(k);
    return raw === null ? fallback : JSON.parse(raw);
  };
  const writeJson = (k, v) => localStorage.setItem(k, JSON.stringify(v));

  let dbPromise = null;
  const openDb = () => {
    if (dbPromise === null) {
      dbPromise = new Promise((resolve, reject) => {
        const req = indexedDB.open(DB_NAME, 1);
        req.onupgradeneeded = () => req.result.createObjectStore(STORE_NAME);
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
      });
    }
    return dbPromise;
  };

  const tx = async (mode, fn) => {
    const db = await openDb();
    return new Promise((resolve, reject) => {
      const t = db.transaction(STORE_NAME, mode);
      const req = fn(t.objectStore(STORE_NAME));
      t.oncomplete = () => resolve(req?.result);
      t.onerror = () => reject(t.error);
    });
  };

  return {
    appendEvent: (e) => {
      const all = readJson(EVENTS_KEY, []);
      all.push(e);
      writeJson(EVENTS_KEY, all);
    },
    readEvents: () => readJson(EVENTS_KEY, []),
    putWord: (w) => {
      const all = readJson(WORDS_KEY, {});
      all[w.id] = w;
      writeJson(WORDS_KEY, all);
    },
    readWords: () => readJson(WORDS_KEY, {}),
    putMeta: (k, v) => {
      const all = readJson(META_KEY, {});
      all[k] = v;
      writeJson(META_KEY, all);
    },
    getMeta: (k) => readJson(META_KEY, {})[k] ?? null,
    putImage: (wordId, blob) => tx('readwrite', (s) => s.put(blob, wordId)),
    getImage: (wordId) => tx('readonly', (s) => s.get(wordId)),
    /** 只保留最近 keep 个词的图；词与事件记录不受影响 */
    pruneImages: async (keep = 20) => {
      const words = readJson(WORDS_KEY, {});
      const ids = Object.values(words)
        .sort((a, b) => b.createdAt - a.createdAt)
        .map((w) => w.id);
      const stale = ids.slice(keep);
      for (const id of stale) await tx('readwrite', (s) => s.delete(id));
      return stale.length;
    },
  };
}
```

// web/units/frame-qc.mjs
// 初值来自设计文档 §5.1，首周用真实数据校准；改动必须记入变更记录。
export const DARK_THRESHOLD = 40;
export const BLUR_THRESHOLD = 80;

export function computeStats(pixels, width, height) {
  const n = width * height;
  let sum = 0;
  for (let i = 0; i < n; i += 1) sum += pixels[i];
  const brightness = sum / n;

  // 4 邻域拉普拉斯，边缘像素跳过
  let lapSum = 0;
  let lapSqSum = 0;
  let count = 0;
  for (let y = 1; y < height - 1; y += 1) {
    for (let x = 1; x < width - 1; x += 1) {
      const i = y * width + x;
      const lap = 4 * pixels[i] - pixels[i - 1] - pixels[i + 1] - pixels[i - width] - pixels[i + width];
      lapSum += lap;
      lapSqSum += lap * lap;
      count += 1;
    }
  }
  if (count === 0) return { brightness, laplacianVar: 0 };
  const mean = lapSum / count;
  return { brightness, laplacianVar: lapSqSum / count - mean * mean };
}

export function judgeFrame({ brightness, laplacianVar }) {
  if (brightness < DARK_THRESHOLD) return { ok: false, reason: 'too_dark' };
  if (laplacianVar < BLUR_THRESHOLD) return { ok: false, reason: 'too_blurry' };
  return { ok: true, reason: 'ok' };
}

// web/units/pick-word.mjs
export function pickWord({ candidates, acceptableSets, exclude = [] }) {
  const accepted = new Set();
  for (const labels of Object.values(acceptableSets)) for (const l of labels) accepted.add(l);

  for (const c of candidates) {
    if (!accepted.has(c.label)) continue;
    if (exclude.includes(c.label)) continue;
    return { word: c.label, reason: `候选 ${c.label} 命中可接受集` };
  }
  return null;
}

// web/units/scheduler.mjs
export const INTERVALS_DAYS = Object.freeze([1, 3, 7]);
const DAY_MS = 86400000;

export function isMaintained(word) {
  return word.maintained === true || word.dueAt === null;
}

export function nextState(word, now) {
  const stage = (word.stage ?? 0) + 1;
  if (stage > INTERVALS_DAYS.length) {
    return { ...word, stage, dueAt: null, maintained: true, lastReviewedAt: now };
  }
  return { ...word, stage, dueAt: now + INTERVALS_DAYS[stage - 1] * DAY_MS, lastReviewedAt: now };
}

// 同刻到期时按 id 的码位序兜底（不用 localeCompare：那取决于 ICU 与运行环境语言）
function compareId(x, y) {
  const a = String(x);
  const b = String(y);
  return a < b ? -1 : a > b ? 1 : 0;
}

export function dueWords(words, now) {
  return Object.values(words)
    .filter((w) => !isMaintained(w) && w.dueAt !== null && w.dueAt <= now)
    .sort((a, b) => (a.dueAt - b.dueAt) || compareId(a.id, b.id));
}

// web/units/feedback.mjs
export const VERDICTS = Object.freeze(['correct', 'flawed', 'uncertain']);
export const ERROR_TYPES = Object.freeze(['word_choice', 'collocation', 'grammar', 'none']);

export function validateFeedback(raw) {
  const errors = [];
  if (raw === null || typeof raw !== 'object') {
    return { ok: false, errors: ['响应必须是对象'] };
  }
  for (const k of ['verdict', 'error_type', 'rewrite', 'note']) {
    if (!(k in raw)) errors.push(`缺少字段 ${k}`);
  }
  if ('verdict' in raw && !VERDICTS.includes(raw.verdict)) {
    errors.push(`verdict 取值越界: ${String(raw.verdict)}`);
  }
  if ('error_type' in raw && !ERROR_TYPES.includes(raw.error_type)) {
    errors.push(`error_type 取值越界: ${String(raw.error_type)}`);
  }
  if ('rewrite' in raw && raw.rewrite !== null) {
    if (typeof raw.rewrite !== 'string' || raw.rewrite.trim() === '') {
      errors.push('rewrite 必须是非空字符串或 null');
    }
  }
  if ('note' in raw && (typeof raw.note !== 'string' || raw.note.trim() === '')) {
    errors.push('note 必须是非空字符串');
  }
  // 强制 verdict 与 error_type 的搭配关系
  if (raw.verdict === 'correct' && raw.error_type !== 'none') {
    errors.push('verdict=correct 时 error_type 必须是 none');
  }
  if (raw.verdict === 'flawed' && raw.error_type === 'none') {
    errors.push('verdict=flawed 时必须指出 error_type');
  }
  return errors.length === 0 ? { ok: true, value: raw } : { ok: false, errors };
}

// server/env.mjs
const REQUIRED = ['DEEPSEEK_API_KEY', 'DEEPSEEK_API_BASE', 'DEEPSEEK_MODEL'];
const OPTIONAL_DEFAULTS = { VISION_DETAIL: 'low' };

export function loadEnv(source = process.env) {
  const missing = REQUIRED.filter((k) => {
    const v = source[k];
    return v === undefined || String(v).trim() === '';
  });
  if (missing.length > 0) {
    // 启动即失败：不静默降级成"没有密钥也能跑"
    throw new Error(`缺少必需的环境变量: ${missing.join(', ')}（见 .env.example）`);
  }
  const env = {};
  for (const k of REQUIRED) env[k] = String(source[k]);
  for (const [k, fallback] of Object.entries(OPTIONAL_DEFAULTS)) {
    const v = source[k];
    env[k] = v === undefined || String(v).trim() === '' ? fallback : String(v);
  }
  env.PORT = Number(source.PORT ?? 8787);
  if (!Number.isInteger(env.PORT) || env.PORT <= 0) {
    throw new Error(`PORT 非法: ${String(source.PORT)}`);
  }
  return env;
}

const json = (res, code, body) => {
  const s = JSON.stringify(body);
  res.writeHead(code, { 'content-type': 'application/json; charset=utf-8' });
  res.end(s);
};

const readBody = (req) => new Promise((resolve, reject) => {
  const chunks = [];
  req.on('data', (c) => chunks.push(c));
  req.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')));
  req.on('error', reject);
});

async function serveStatic(req, res) {
  const url = new URL(req.url, 'http://localhost');
  const rel = url.pathname === '/' ? '/index.html' : url.pathname;
  const full = join(WEB_ROOT, normalize(rel).replace(/^([/\\])+/, ''));
  if (!full.startsWith(WEB_ROOT)) return json(res, 403, { error: 'forbidden' });
  try {
    const buf = await readFile(full);
    res.writeHead(200, { 'content-type': MIME[extname(full)] ?? 'application/octet-stream' });
    res.end(buf);
  } catch {
    json(res, 404, { error: 'not_found' });
  }
}

const server = createServer(async (req, res) => {
  if (req.method === 'POST' && req.url === '/api/recognize') {
    // 密钥只在服务端注入；客户端永远拿不到
    return json(res, 200, { ok: false, error: 'not_implemented_until_task_8' });
  }
  if (req.method === 'POST' && req.url === '/api/feedback') {
    return json(res, 200, { ok: false, error: 'not_implemented_until_task_8' });
  }
  if (req.method === 'GET' || req.method === 'HEAD') return serveStatic(req, res);
  return json(res, 405, { error: 'method_not_allowed' });
});

// web/app.mjs
// 状态机为纯逻辑：不引用任何浏览器 API，便于在 Node 中直接测试。
// DOM 渲染与副作用在 mount() 中，测试不触碰。
const TRANSITIONS = {
  ready: { capture: 'capturing' },
  capturing: { frameOk: 'word', frameBad: 'ready' },
  word: { wordReady: 'reading' },
  reading: { readDone: 'composing', skipReading: 'composing' },
  composing: { submit: 'feedback' },
  feedback: { rewrite: 'composing', next: 'done' },
  done: {},
};

export function createMachine({ onEnter }) {
  let state = 'ready';
  const ctx = { rewriteCount: 0, frameRejections: 0, skippedReading: false };
  onEnter(state);

  return {
    get state() { return state; },
    can: (action) => Boolean(TRANSITIONS[state]?.[action]),
    snapshot: () => ({ state, ...ctx }),
    send(action) {
      const next = TRANSITIONS[state]?.[action];
      if (next === undefined) return false; // 非法动作静默忽略，不改状态
      if (action === 'frameBad') ctx.frameRejections += 1;
      if (action === 'skipReading') ctx.skippedReading = true;
      if (action === 'submit') ctx.rewriteCount += 1;
      state = next;
      onEnter(state);
      return true;
    },
  };
}

    video: { facingMode, width: { ideal: 1280 } },

/** 从 video 取一帧，返回 { blob, stats } —— stats 供 frameQC 判定 */
export async function grabFrame(videoEl, canvasEl, { maxEdge = 512 } = {}) {
  const vw = videoEl.videoWidth;
  const vh = videoEl.videoHeight;
  const scale = Math.min(1, maxEdge / Math.max(vw, vh));
  const w = Math.round(vw * scale);
  const h = Math.round(vh * scale);
  canvasEl.width = w;
  canvasEl.height = h;
  const ctx = canvasEl.getContext('2d');
  ctx.drawImage(videoEl, 0, 0, w, h);
  const blob = await new Promise((r) => canvasEl.toBlob(r, 'image/jpeg', 0.8));
  return { blob, pixels: ctx.getImageData(0, 0, w, h).data, width: w, height: h };
}

import { judgeFrame } from './frame-qc.mjs';
import { pickWord } from './pick-word.mjs';

export async function recognize(blob, { fetchImpl = fetch } = {}) {
  const form = new FormData();
  form.append('image', blob, 'frame.jpg');
  const res = await fetchImpl('/api/recognize', { method: 'POST', body: form });
  if (!res.ok) throw new Error(`识物请求失败 HTTP ${res.status}`);
  const data = await res.json();
  if (data.ok !== true || !Array.isArray(data.candidates)) {
    throw new Error('识物响应结构非法');
  }
  return { candidates: data.candidates };
}

/**
 * 最多尝试 2 次（用户第 2 次仍失败 -> manual）。
 * 刻意不做自动连拍兜底：retry_rate 是正式闸门，必须保持可测。
 */
export async function recognizeWithFallback({ grab, acceptableSets, exclude, fetchImpl }) {
  let attempts = 0;
  for (let i = 0; i < 2; i += 1) {
    const { blob, stats } = await grab();
    const verdict = judgeFrame(stats);
    if (!verdict.ok) return { mode: 'frame_rejected', reason: verdict.reason, word: null, attempts };
    attempts += 1;
    try {
      const { candidates } = await recognize(blob, { fetchImpl });
      const picked = pickWord({ candidates, acceptableSets, exclude });
      if (picked !== null) return { mode: 'ok', word: picked.word, candidates, attempts };
    } catch {
      // 落空后继续下一轮；两轮都失败即降级
    }
  }
  return { mode: 'manual', word: null, attempts };
}

// web/units/compose.mjs
import { validateFeedback } from './feedback.mjs';

export async function submitSentence({ sentence, word, scene }, { fetchImpl = fetch } = {}) {
  if (typeof sentence !== 'string' || sentence.trim() === '') {
    return { status: 'pending', error: 'empty_sentence', sentence };
  }
  try {
    const res = await fetchImpl('/api/feedback', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ sentence, word, scene }),
    });
    if (!res.ok) return { status: 'pending', error: `http_${res.status}`, sentence };
    const raw = await res.json();
    const v = validateFeedback(raw);
    if (!v.ok) return { status: 'pending', error: v.errors.join('; '), sentence };
    return { status: 'ok', feedback: v.value, uncertain: v.value.verdict === 'uncertain', sentence };
  } catch (err) {
    return { status: 'pending', error: String(err?.message ?? err), sentence };
  }
}
```

// 只判"有没有说出目标词"，不做音素级音准评分（设计文档 §4.3 的显式妥协）。
export function checkSpeech(targetWord, transcript) {

  const w = String(targetWord).toLowerCase();
  if (w === '' || t === '') return { said: false, transcript: String(transcript ?? '') };

// scripts/export.mjs
const rate = (num, den) => (den === 0 ? 0 : num / den);

export function summarize(events, words) {
  const bySession = new Map();
  for (const e of events) {
    if (!bySession.has(e.sessionId)) bySession.set(e.sessionId, { rejects: 0 });
    if (e.type === 'frame_rejected') bySession.get(e.sessionId).rejects += 1;
  }
  const sessions = [...bySession.values()];
  const retried = sessions.filter((s) => s.rejects >= 2).length;

  const count = (t) => events.filter((e) => e.type === t).length;
  const feedbackTotal = count('feedback_ok') + count('feedback_pending') + count('uncertain');

  return {
    sessions: sessions.length,
    retryRate: rate(retried, sessions.length),
    uncertaintyRate: rate(count('uncertain'), feedbackTotal),
    frameRejected: count('frame_rejected'),
    recognizeFailed: count('recognize_failed'),
    speechUnsupported: count('speech_unsupported'),
    manualRecurrence: count('recurrence_manual'),
    sceneRecurrence: count('recurrence_scene'),
    pendingCount: count('feedback_pending'),
    wordsTracked: Object.keys(words ?? {}).length,
  };
}

const escapeCell = (v) => {
  const s = v === null || v === undefined ? '' : String(v);
  return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
};

export function toCsv(events) {
  const header = ['ts', 'type', 'sessionId', 'wordId', 'payload'];
  const rows = events.map((e) => [
    e.ts, e.type, e.sessionId, e.wordId ?? '', JSON.stringify(e.payload ?? {}),
  ]);
  return [header, ...rows].map((r) => r.map(escapeCell).join(',')).join('\n') + '\n';
}
```
